-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Profiles table (extends Supabase auth.users)
CREATE TABLE profiles (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL UNIQUE,
    email TEXT NOT NULL,
    full_name TEXT NOT NULL,
    phone TEXT,
    avatar_url TEXT,
    role TEXT NOT NULL CHECK (role IN ('client', 'professional', 'admin', 'super_admin')),
    status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'rejected', 'suspended')),
    preferred_language TEXT DEFAULT 'en' CHECK (preferred_language IN ('en', 'si', 'ta')),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Professional profiles

    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    profile_id UUID REFERENCES profiles(id) ON DELETE CASCADE NOT NULL,

    specialization TEXT NOT NULL,
    qualifications TEXT[] DEFAULT '{}',

# Hopes.lk Mobile Application

A production-ready, trilingual (Sinhala/English/Tamil) mobile platform connecting Sri Lankan users with verified professionals for online consultations.

## 🚀 Tech Stack

- **Framework**: React Native + Expo (SDK 51)
- **Language**: TypeScript (Strict)
- **Backend**: Supabase (PostgreSQL + Auth + Realtime)
- **Navigation**: Expo Router (File-based)
- **State**: Zustand + React Context
- **Storage**: Expo Secure Store (Encrypted)
- **UI**: Custom component library with dark mode support
- **Payments**: Server-side architecture ready for gateway integration

## 📁 Project Structure
app/ # Expo Router screens (auth)/ # Auth group (login, register, forgot-password) (client)/ # Client screens (home, explore, appointments, resources, profile) (professional)/ # Professional dashboard (admin)/ # Admin panel components/ ui/ # Reusable UI primitives layout/ # SafeArea, Header, BottomNav cards/ # ProfessionalCard, CategoryCard, AppointmentCard contexts/ # Auth, Theme, Language providers hooks/ # Data fetching hooks with Supabase services/ # API service layer types/ # TypeScript definitions locales/ # Translation files (en, si, ta) lib/ # Supabase client supabase/migrations/ # Database schema, RLS policies, seed data
